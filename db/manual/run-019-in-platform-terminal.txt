// AIC platform migration 019. Run inside the aic-platform container; before or after the deploy both work:
//   paste this whole file into Coolify -> aic-platform -> Terminal.
import { createRequire } from 'node:module';
const require = createRequire('/app/apps/platform/server.js');
let pg;
try { pg = require('pg'); } catch {
  try { pg = createRequire(import.meta.url)('pg'); } catch {
    console.error("The 'pg' package is not in this container. Run this in the platform app's terminal, not the database's.");
    process.exit(1);
  }
}
const SQL = `
-- 019_onboarding_link_emails.sql
--
-- Client onboarding links are emailed by AIC itself, from aiccertified.cloud,
-- rather than pasted into someone's own email, so the client receives them
-- from a recognisable sender. Each send is recorded: to whom, by whom, when,
-- and whether the mail service accepted it.
--
-- AIC's own table: no row-level policy, withheld from aic_tenant.
-- Additive and safe to run more than once.

CREATE TABLE IF NOT EXISTS client_onboarding_link_emails (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  link_id uuid NOT NULL REFERENCES client_onboarding_links(id) ON DELETE CASCADE,
  sent_to varchar(255) NOT NULL,
  sent_by uuid REFERENCES users(id) ON DELETE SET NULL,
  accepted boolean NOT NULL,
  failure varchar(40),
  sent_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX IF NOT EXISTS client_onboarding_link_emails_link_idx ON client_onboarding_link_emails (link_id, sent_at DESC);

DO $outer$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'aic_tenant') THEN
    EXECUTE 'REVOKE ALL ON client_onboarding_link_emails FROM aic_tenant';
  END IF;
END $outer$;
`;
const url = process.env.DATABASE_URL || process.env.POSTGRES_URL;
if (!url) { console.error('No DATABASE_URL in this container.'); process.exit(1); }
const client = new pg.Client({ connectionString: url });
client.on('notice', (n) => console.log('   · ' + n.message));
await client.connect();
process.stdout.write('019 onboarding link emails ... ');
try {
  await client.query('BEGIN'); await client.query(SQL); await client.query('COMMIT'); console.log('OK');
} catch (e) {
  await client.query('ROLLBACK').catch(() => {});
  console.log('FAILED — rolled back, nothing was applied'); console.log('   ' + e.message); await client.end(); process.exit(1);
}
const ok = (await client.query(`SELECT to_regclass('public.client_onboarding_link_emails') IS NOT NULL AS ok`)).rows[0].ok;
console.log(ok ? '\u2713 019 verified. Now push or redeploy the platform.' : '\u2717 019 not complete. Send the output above to Claude.');
await client.end();
AIC_END
