cat > /tmp/fix-tenant.mjs <<'AIC_END'
// Sets aic_tenant's password to exactly what TENANT_DATABASE_URL contains, then tests the login.
// Run inside the aic-platform container: it uses DATABASE_URL (the owner) to make the change. Prints no secrets.
import { createRequire } from 'node:module';
const require = createRequire('/app/apps/platform/server.js');
const pg = require('pg');
const tUrl = process.env.TENANT_DATABASE_URL, oUrl = process.env.DATABASE_URL;
if (!tUrl || !oUrl) { console.log('Missing TENANT_DATABASE_URL or DATABASE_URL in this container'); process.exit(1); }
const u = new URL(tUrl);
const pw = decodeURIComponent(u.password);
console.log(`user=${u.username} host=${u.hostname} db=${u.pathname.slice(1)} password length=${pw.length} hex-only=${/^[0-9a-f]+$/i.test(pw)}`);
const test = async () => { const c = new pg.Client({ connectionString: tUrl }); try { await c.connect(); const r = await c.query('select current_user'); await c.end(); return 'OK as ' + r.rows[0].current_user; } catch (e) { return 'FAIL: ' + e.message; } };
console.log('Before:', await test());
const o = new pg.Client({ connectionString: oUrl });
await o.connect();
await o.query(`ALTER ROLE aic_tenant WITH LOGIN PASSWORD ${o.escapeLiteral(pw)}`);
await o.end();
console.log('Password set from TENANT_DATABASE_URL.');
console.log('After:', await test());
AIC_END
node /tmp/fix-tenant.mjs
